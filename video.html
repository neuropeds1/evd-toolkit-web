function youtubeId(url) {
  if (!url) return null;
  const patterns = [
    /youtu\.be\/([a-zA-Z0-9_-]+)/,
    /youtube\.com\/watch\?v=([a-zA-Z0-9_-]+)/,
    /youtube\.com\/embed\/([a-zA-Z0-9_-]+)/
  ];
  for (const p of patterns) {
    const m = url.match(p);
    if (m) return m[1];
  }
  return null;
}
// Deterministic color per category name — same category always
// gets the same color, no manual mapping to maintain.
function categoryColor(name) {
  const palette = ["#2c7be5", "#27ae60", "#e67e22", "#8e44ad", "#e74c3c", "#16a2b8", "#d63384", "#6c757d"];
  let hash = 0;
  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
  return palette[Math.abs(hash) % palette.length];
}

  const container = html`<div class="fs-filters"></div>`;

    container.innerHTML = "";
    videoCategories.forEach(cat => {
      const isActive = cat === container.value;
      const color = cat === "All" ? "#2c7be5" : categoryColor(cat);
      const pill = html`<button type="button" class="fs-filter-pill${isActive ? " active" : ""}">${cat}</button>`;
      if (isActive) {
        pill.style.background = color;
        pill.style.borderColor = color;
      }
      pill.onclick = () => {
        container.value = cat;
        container.dispatchEvent(new CustomEvent("input"));
        draw();
      };
      container.appendChild(pill);
    });
  }
  draw();
  return container;
}
