  const app = getApps().length ? getApp() : initializeApp(firebaseConfig);
  const db = getFirestore(app);

  return Generators.observe(notify => {
    let textItems = [];
    let resources = [];
    const emit = () => notify({ textItems, resources });

    const unsubText = onSnapshot(collection(db, "DownloadsText"), snapshot => {
      textItems = snapshot.docs.map(d => d.data());
      emit();
    });
    const unsubResources = onSnapshot(collection(db, "Downloads"), snapshot => {
      resources = snapshot.docs.map(d => d.data()).sort((a, b) => (a.order ?? 0) - (b.order ?? 0));
      emit();
    });

    return () => { unsubText(); unsubResources(); };
  });
}
overviewParagraphs = downloadsData.textItems.filter(t => t.section === "overview").sort((a, b) => (a.order ?? 0) - (b.order ?? 0))
futureDirections = downloadsData.textItems.filter(t => t.section === "futureDirections").sort((a, b) => (a.order ?? 0) - (b.order ?? 0))
