// Firestore only stores pmid/type/color — fetch title/authors/journal
// live from PubMed's own API (NCBI E-utilities), batched into one request.
referencesWithMeta = {
  const pmids = references.map(r => r.pmid).filter(Boolean);
  if (pmids.length === 0) return references;

  try {
    const url = `https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=pubmed&id=${pmids.join(",")}&retmode=json`;
    const res = await fetch(url);
    const data = await res.json();

    return references.map(r => {
      const s = data.result?.[r.pmid];
      if (!s) return r;
      return {
        ...r,
        title: s.title,
        authors: (s.authors || []).map(a => a.name).join(", "),
        journal: s.fulljournalname,
        pubdate: s.pubdate
      };
    });
  } catch (e) {
    return references;
  }
}
groupedReferences = {
  const groups = new Map();
  referencesWithMeta.forEach(r => {
    const key = r.type ?? "Uncategorized";
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(r);
  });
  return [...groups.entries()].sort((a, b) => a[0].localeCompare(b[0]));
}
// Deterministic color per category name — same category always
// gets the same color, no manual mapping to maintain.
function categoryColor(name) {
  const palette = ["#007AFF", "#34C759", "#FF9500", "#AF52DE", "#FF3B30", "#30B0C7", "#FF2D55", "#8E8E93"];
  let hash = 0;
  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
  return palette[Math.abs(hash) % palette.length];
}
