// Semantic color, not arbitrary: antimicrobial-impregnated catheters
// are the preferred choice (green); non-antimicrobial is red as a
// visual nudge toward the preferred option. Check the "non-" case
// FIRST since "Non-antimicrobial" also contains the substring
// "antimicrobial" and would otherwise false-match as green.
// Other coating categories (e.g. antithrombogenic) sit on a different
// axis entirely -- not "avoid", just a different property -- so they
// get a neutral blue rather than falling into the same red bucket.
function categoryColor(name) {
  const lower = name.toLowerCase();
  if (/non[\s-]?antimicrobial/.test(lower)) return "#e74c3c";
  if (/antimicrobial/.test(lower)) return "#27ae60";
  if (/antithrombogenic|thrombo/.test(lower)) return "#2c7be5";
  return "#8e44ad";
}
// Curated clinical pearls per catheter category. Add more entries here

categoryPearls = ({

    "Commonly coated with agents such as rifampicin and clindamycin, or minocycline and rifampin.",
    "Device choice is one element of infection prevention — full bundle compliance (sterile technique, minimizing access, etc.) still matters most."
  ]
})
groupedTypes = {
  const groups = new Map();
  catheterTypes.forEach(t => {
    const key = t.category ?? "Uncategorized";
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(t);
  });
  return [...groups.entries()].sort((a, b) => a[0].localeCompare(b[0]));
}
