// Group steps by category, sorted by categoryOrder then step order --
// mirrors MRISafetyViewModel's `group()` function in the app.
groupedSteps = {
  const groups = new Map();
  mriData.steps.forEach(s => {
    const key = s.category ?? "Uncategorized";
    if (!groups.has(key)) groups.set(key, { categoryOrder: s.categoryOrder ?? 999, steps: [] });
    groups.get(key).steps.push(s);
  });
  return [...groups.entries()]
    .map(([category, { categoryOrder, steps }]) => [category, categoryOrder, steps.sort((a, b) => (a.order ?? 0) - (b.order ?? 0))])
    .sort((a, b) => a[1] - b[1]);
}
function categoryColor(name) {
  const palette = ["#2c7be5", "#27ae60", "#e67e22", "#8e44ad", "#e74c3c", "#16a2b8", "#d63384", "#6c757d"];
  let hash = 0;
  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
  return palette[Math.abs(hash) % palette.length];
}
