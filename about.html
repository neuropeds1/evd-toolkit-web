// Port of AboutView.swift's people-sorting rule: ordered items first
// (by order), then unordered items alphabetically by name.
sortPeople = function(people) {
  const ordered = people.filter(p => p.order != null).sort((a, b) => a.order - b.order);
  const unordered = people.filter(p => p.order == null).sort((a, b) => (a.name ?? "").localeCompare(b.name ?? ""));
  return [...ordered, ...unordered];
}
isDeveloper = c => (c.Role ?? "").toLowerCase().includes("design")
developers = sortPeople(collaborators.filter(isDeveloper))
collaboratorsList = sortPeople(collaborators.filter(c => !isDeveloper(c)))

}
