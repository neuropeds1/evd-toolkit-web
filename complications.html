Complications

Recognize complications early. Understand patterns. Prevent avoidable harm.

TipNormal / Expected Position

Target: Frontal horn near foramen of Monro.

  • Tip should float freely in CSF
  • Avoid ventricular wall contact
  • Confirm on CT after placement

// Deterministic color per category name — same category always
// gets the same color, no manual mapping to maintain.
function categoryColor(name) {
  const palette = ["#2c7be5", "#27ae60", "#e67e22", "#8e44ad", "#e74c3c", "#16a2b8", "#d63384", "#6c757d"];
  let hash = 0;
  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
  return palette[Math.abs(hash) % palette.length];
}

groupedComplications = {
  const groups = new Map();
  complications.forEach(c => {
    const key = c.category ?? "Uncategorized";
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(c);
  });
  return [...groups.entries()].sort((a, b) => a[0].localeCompare(b[0]));
