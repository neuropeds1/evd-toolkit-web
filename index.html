// Port of ContentView.swift's todaysPearl: same pearl for everyone on a
// given calendar day, rotating through the live Firestore-backed list.
todaysPearl = {
  if (pearlsData.length === 0) return null;
  const start = new Date(new Date().getFullYear(), 0, 0);
  const diff = new Date() - start;
  const dayOfYear = Math.floor(diff / (1000 * 60 * 60 * 24));
  return pearlsData[dayOfYear % pearlsData.length];
}
todaysPearl
  ? html`<div class="ios-card pearl-card" style="--card-color: var(--ios-blue);">

    </div>`
